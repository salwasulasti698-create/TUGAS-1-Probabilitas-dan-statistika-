#include <iostream>
#include <cmath>
using namespace std;

int main() {
    int n = 5;
    double p = 0.60;
    double q = 1 - p;
    double hasil = 0;

    for (int k = 3; k <= 5; k++) {
        double kombinasi = 1;

        for (int i = 1; i <= k; i++) {
            kombinasi *= (n - i + 1.0) / i;
        }

        hasil += kombinasi * pow(p, k) * pow(q, n-k);
    }

    cout << "Peluang prediksi benar = " << hasil << endl;
    cout << "Persentase = " << hasil * 100 << "%" << endl;

    return 0;
}
